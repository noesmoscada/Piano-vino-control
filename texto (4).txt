<div id="contenedor"></div>


<script>

const contenedor = document.getElementById("contenedor");


for(let i=1;i<=150;i++){

let codigo="PV-"+String(i).padStart(4,"0");


let caja=document.createElement("div");

caja.className="qr";


caja.innerHTML="<h3>"+codigo+"</h3>";


let qr=document.createElement("div");


caja.appendChild(qr);


contenedor.appendChild(caja);


new QRCode(qr,{
text:codigo,
width:180,
height:180
});


}

</script>
